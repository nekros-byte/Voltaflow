    function calculateMSEDCLBill(units) {
      const u = Math.max(0, Number(units) || 0);

      // Telescopic Energy Charges
      const s1Units = Math.min(u, 100);
      const s2Units = Math.min(Math.max(u - 100, 0), 200);
      const s3Units = Math.min(Math.max(u - 300, 0), 200);
      const s4Units = Math.max(u - 500, 0);

      const s1Cost = s1Units * 4.71;
      const s2Cost = s2Units * 10.29;
      const s3Cost = s3Units * 14.55;
      const s4Cost = s4Units * 16.64;

      const energyCharge = s1Cost + s2Cost + s3Cost + s4Cost;
      const fixedCharge = 128.0; // Single-Phase LT-1 Residential
      const wheelingCharge = u * 1.17;
      const facCharge = u * 0.65;

      const subTotal = energyCharge + fixedCharge + wheelingCharge + facCharge;
      const electricityDuty = subTotal * 0.16; // 16% Maharashtra Govt Duty
      const total = subTotal + electricityDuty;

      // Determine active marginal slab
      let activeSlabName = 'Slab 1';
      let activeSlabRate = 4.71;
      if (u > 500) {
        activeSlabName = 'Slab 4';
        activeSlabRate = 16.64;
      } else if (u > 300) {
        activeSlabName = 'Slab 3';
        activeSlabRate = 14.55;
      } else if (u > 100) {
        activeSlabName = 'Slab 2';
        activeSlabRate = 10.29;
      }

      return {
        units: u,
        s1Cost,
        s2Cost,
        s3Cost,
        s4Cost,
        energyCharge,
        fixedCharge,
        wheelingCharge,
        facCharge,
        electricityDuty,
        total,
        activeSlabName,
        activeSlabRate
      };
    }

    // Marginal daily cost estimate based on current accrued tariff slab
    function estimateDailyCost(dayKwh, marginalRate) {
      const effectivePerUnit = marginalRate + 1.17 + 0.65;
      return dayKwh * effectivePerUnit * 1.16;
    }

    function getProjectedUnits() {

    }

    // GOOGLE SHEETS CSV / APPS SCRIPT JSON PARSER & FETCH
    // -------------------------------------------------------------------------
    function parseSheetTextToDays(rawText) {
      const trimmed = rawText.trim();

      // 1. Try JSON (Apps Script Web App Endpoint)
      if (trimmed.startsWith('[') || trimmed.startsWith('{')) {
        const json = JSON.parse(trimmed);
        const rows = Array.isArray(json) ? json : (json.data || json.rows || []);
        if (!rows.length) throw new Error('Empty JSON array');

        return rows.slice(-7).map((r, idx) => {
          const kwhVal = parseFloat(r.Units_kWh ?? r.units_kwh ?? r.kWh ?? r.kwh ?? r.Units ?? 0);
          const peakVal = parseInt(r.Peak_W ?? r.peak_w ?? r.Peak ?? 1350, 10);
          const dateStr = String(r.Date ?? r.date ?? `2026-11-${10 + idx}`);
          const dayNames = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
          const dayStr = String(r.Day ?? r.day ?? dayNames[idx % 7]).slice(0, 3);
          return {
            date: dateStr,
            day: dayStr,
            kwh: isNaN(kwhVal) ? 8.5 : Number(kwhVal.toFixed(1)),
            peakW: isNaN(peakVal) ? 1350 : peakVal
          };
        });
      }

      // 2. Parse CSV (Published Google Sheet ?output=csv)
      const lines = trimmed.split(/\r?\n/).filter(l => l.trim().length > 0);
      if (lines.length < 2) throw new Error('CSV needs header and data rows');

      const headers = lines[0].split(',').map(h => h.replace(/"/g, '').trim().toLowerCase());
      const dateIdx = headers.findIndex(h => h.includes('date'));
      const dayIdx = headers.findIndex(h => h === 'day' || h.includes('weekday'));
      const kwhIdx = headers.findIndex(h => h.includes('kwh') || h.includes('unit'));
      const peakIdx = headers.findIndex(h => h.includes('peak') || h.includes('watt') || h === 'w');

      if (kwhIdx === -1) {
        throw new Error('Could not find Units_kWh or kWh column');
      }

      const dayNames = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
      const dataLines = lines.slice(1).slice(-7);

      return dataLines.map((line, idx) => {
        const cols = line.split(',').map(c => c.replace(/"/g, '').trim());
        const kwhVal = parseFloat(cols[kwhIdx]);
        const peakVal = peakIdx !== -1 ? parseInt(cols[peakIdx], 10) : 1380;
        const dateStr = dateIdx !== -1 && cols[dateIdx] ? cols[dateIdx] : `2026-11-${10 + idx}`;
        const dayStr = dayIdx !== -1 && cols[dayIdx] ? cols[dayIdx].slice(0, 3) : dayNames[idx % 7];

        return {
          date: dateStr,
          day: dayStr,
          kwh: isNaN(kwhVal) ? 8.0 : Number(kwhVal.toFixed(1)),
          peakW: isNaN(peakVal) ? 1380 : peakVal
        };
      });
    }

    async function fetchFromGoogleSheet() {
      const urlInput = document.getElementById('sheetUrlInput').value.trim();
      if (!urlInput) {
        showToast('Paste a Google Sheet CSV URL first');
        return;
      }

      const btn = document.getElementById('fetchSheetBtn');
      btn.textContent = 'Syncing...';
      btn.disabled = true;

      try {
        const response = await fetch(urlInput);
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const text = await response.text();
        const parsedDays = parseSheetTextToDays(text);

        if (parsedDays.length > 0) {
          appState.days = parsedDays;
          appState.sourceName = 'Live_Google_Sheet';
          appState.selectedDayIndex = parsedDays.length - 1;
          document.getElementById('syncStatusText').textContent = 'Google Sheet · Synced just now';
          updateDashboardUI();
          renderSheetPreviewTable();
          showToast(`Synced ${parsedDays.length} rows from Google Sheet!`);
          closeSheetModal();
        }
      } catch (err) {
        console.error(err);
        showToast('CORS/URL error — verify "Publish to web (CSV)"');
      } finally {
        btn.textContent = 'Fetch Live Sheet';
        btn.disabled = false;
      }
    }

            <td class="py-1.5 font-bold">${d.day}</td>
            <td class="py-1.5 font-bold text-[#2B7FFF]">${d.kwh.toFixed(1)}</td>

    function renderTariffModal() {
      const accruedUnits = getAccruedUnits();
      const projectedUnits = getProjectedUnits();

      document.getElementById('tabAccruedUnits').textContent = accruedUnits.toFixed(0);
      document.getElementById('tabProjectedUnits').textContent = projectedUnits;

      const accBtn = document.getElementById('billTabAccrued');
      const projBtn = document.getElementById('billTabProjected');

      if (appState.billModalMode === 'accrued') {
        accBtn.className = 'py-1.5 rounded-xl text-xs font-bold bg-[#FF6B5B] text-white transition-all';
        projBtn.className = 'py-1.5 rounded-xl text-xs font-bold text-[#6E717C] transition-all';
      } else {
        projBtn.className = 'py-1.5 rounded-xl text-xs font-bold bg-[#FF6B5B] text-white transition-all';
        accBtn.className = 'py-1.5 rounded-xl text-xs font-bold text-[#6E717C] transition-all';
      }

      const targetUnits = appState.billModalMode === 'accrued' ? accruedUnits : projectedUnits;
      const bill = calculateMSEDCLBill(targetUnits);

      document.getElementById('lineSlab1').textContent = `₹${bill.s1Cost.toFixed(2)}`;
      document.getElementById('lineSlab2').textContent = `₹${bill.s2Cost.toFixed(2)}`;
      document.getElementById('lineSlab3').textContent = `₹${bill.s3Cost.toFixed(2)}`;
      document.getElementById('lineSlab4').textContent = `₹${bill.s4Cost.toFixed(2)}`;
      document.getElementById('lineFixed').textContent = `₹${bill.fixedCharge.toFixed(2)}`;
      document.getElementById('lineWheeling').textContent = `₹${bill.wheelingCharge.toFixed(2)}`;
      document.getElementById('lineFac').textContent = `₹${bill.facCharge.toFixed(2)}`;
      document.getElementById('lineDuty').textContent = `₹${bill.electricityDuty.toFixed(2)}`;
      document.getElementById('lineGrandTotal').textContent = `₹${Math.round(bill.total).toLocaleString('en-IN')}`;

      const remainingDays = Math.max(1, appState.totalCycleDays - appState.cycleDay);
      const safeAllowance = Math.max(0, (300 - accruedUnits) / remainingDays).toFixed(1);
      const adviceBox = document.getElementById('slabAdviceBox');

      if (projectedUnits > 300) {
        adviceBox.innerHTML = `<strong class="text-[#FF6B5B]">Slab 3 Breach Alert:</strong> Projected usage (${projectedUnits} kWh) crosses the 300-unit mark. Keep daily consumption under <strong>${safeAllowance} kWh/day</strong> for the remaining ${remainingDays} days to avoid the ₹14.55/unit slab.`;
      } else {
        adviceBox.innerHTML = `<strong class="text-[#2B7FFF]">On Track for Safe Slab:</strong> You are projected to finish at <strong>${projectedUnits} kWh</strong>, safely within Slab 2 (≤300 units).`;
      }
    }

    function updateDashboardUI() {
      const accruedUnits = getAccruedUnits();
      const projectedUnits = getProjectedUnits();
      const accruedBill = calculateMSEDCLBill(accruedUnits);
      const projectedBill = calculateMSEDCLBill(projectedUnits);
      const selectedDay = appState.days[appState.selectedDayIndex] || appState.days[0];

      // Header Cycle & Weekly Summary
      document.getElementById('cycleDayText').textContent = appState.cycleDay;
      const weekTotal = appState.days.reduce((s, d) => s + d.kwh, 0).toFixed(1);
      document.getElementById('weeklySummaryMetric').textContent = `${weekTotal} kWh this wk • Pune LT-1`;

      // Left Hero Card: Circular Ring Progress (Accrued kWh vs 300 kWh Safe Slab Limit)
      const pct = Math.min(100, Math.round((accruedUnits / appState.safeMonthlyTarget) * 100));
      const offset = RING_CIRCUMFERENCE - (pct / 100) * RING_CIRCUMFERENCE;
      document.getElementById('unitsRingArc').style.strokeDashoffset = offset.toFixed(2);
      document.getElementById('accruedUnitsText').textContent = accruedUnits.toFixed(0);
      document.getElementById('safeLimitUnitsText').textContent = appState.safeMonthlyTarget;
      document.getElementById('slabBudgetPct').textContent = `${pct}%`;
      document.getElementById('accruedBillBadge').textContent = `₹${Math.round(accruedBill.total).toLocaleString('en-IN')} so far`;
      document.getElementById('miniProjectedBill').textContent = `₹${Math.round(projectedBill.total).toLocaleString('en-IN')}`;

      // Right Hero Card: Eco-Saving Streak & Eco Guard Button
      document.getElementById('streakCountText').textContent = appState.streak;
      document.getElementById('bestStreakText').textContent = `${appState.bestStreak} days`;

      const ecoBtn = document.getElementById('ecoModeBtn');
      const ecoLabel = document.getElementById('ecoModeLabel');
      if (appState.ecoGuardActive) {
        ecoBtn.className = 'tap-scale w-full py-2 px-2 rounded-xl text-[11px] font-bold flex items-center justify-center gap-1.5 bg-[#FF6B5B] text-white shadow-sm shadow-[#FF6B5B]/30';
        ecoLabel.textContent = 'Eco Active (-15%)';
      } else {
        ecoBtn.className = 'tap-scale w-full py-2 px-2 rounded-xl text-[11px] font-bold flex items-center justify-center gap-1.5 bg-[#FFF0EE] text-[#FF6B5B] border border-[#FF6B5B]/20';
        ecoLabel.textContent = 'Eco Guard';
      }

      // Forecast & Tariff Pills
      document.getElementById('pillProjectedCost').textContent = `₹${Math.round(projectedBill.total).toLocaleString('en-IN')}`;
      const projNote = document.getElementById('pillProjectedUnits');
      if (projectedUnits > 300) {
        projNote.textContent = `~${projectedUnits} kWh · >300 Slab!`;
        projNote.className = 'text-[9px] font-bold text-[#FF6B5B] block';
      } else {
        projNote.textContent = `~${projectedUnits} kWh · Safe Slab`;
        projNote.className = 'text-[9px] font-bold text-emerald-600 block';
      }

      document.getElementById('pillActiveSlabRate').innerHTML = `₹${accruedBill.activeSlabRate.toFixed(2)}<span class="text-[10px] font-medium text-[#6E717C]">/u</span>`;
      document.getElementById('pillActiveSlabName').textContent = `${accruedBill.activeSlabName} · Tap Bill`;

      // Selected Day Caption above Bar Chart
      const selectedEstCost = Math.round(estimateDailyCost(selectedDay.kwh, accruedBill.activeSlabRate));
      document.getElementById('selectedDayDetail').textContent =
        `${selectedDay.day} · ${selectedDay.kwh.toFixed(1)} kWh (₹${selectedEstCost} est. · Peak ${selectedDay.peakW.toLocaleString()}W)`;

      renderDaySelector();
      renderEnergyBars();
    }

    function selectDay(index) {
      appState.selectedDayIndex = index;

    function logTestUnits() {
      const targetDay = appState.days[appState.selectedDayIndex];
      targetDay.kwh = Number((targetDay.kwh + 2.0).toFixed(1));
      targetDay.peakW = Math.min(3500, targetDay.peakW + 240);
      updateDashboardUI();
      showToast(`Logged +2.0 kWh on ${targetDay.day} (${targetDay.kwh} kWh)`);
    }

    function toggleEcoMode() {
      appState.ecoGuardActive = !appState.ecoGuardActive;
      if (appState.ecoGuardActive) {
        appState.streak += 1;
        if (appState.streak > appState.bestStreak) appState.bestStreak = appState.streak;
        showToast('Eco Guard ON: Forecasting 15% lower daily load');
      } else {
        appState.streak = Math.max(0, appState.streak - 1);
        showToast('Eco Guard paused — baseline forecast restored');
      }
      updateDashboardUI();
    }

    function setChartMode(mode) {
      appState.chartMode = mode;
      const kwhBtn = document.getElementById('modeKwhBtn');
      const costBtn = document.getElementById('modeCostBtn');

      if (mode === 'kwh') {
        kwhBtn.className = 'px-2.5 py-1 rounded-lg bg-[#2B7FFF] text-white shadow-xs transition-all';
        costBtn.className = 'px-2.5 py-1 rounded-lg text-[#6E717C] hover:text-[#1C1D22] transition-all';
      } else {
        costBtn.className = 'px-2.5 py-1 rounded-lg bg-[#2B7FFF] text-white shadow-xs transition-all';
        kwhBtn.className = 'px-2.5 py-1 rounded-lg text-[#6E717C] hover:text-[#1C1D22] transition-all';
      }
      renderEnergyBars();
    }

    }
